import pandas as pd

def create_dataframe():
    # Create a DataFrame
    data = {
        'Column1': [1, 2, 3],
        'Column2': ['A', 'B', 'C'],
        'Column3': [4.5, 5.5, 6.5]
    }
    return pd.DataFrame(data)

def extract_columns(df):
    # Extract specific columns
    extracted_columns = df[['Column1', 'Column2']]
    return extracted_columns

if __name__ == "__main__":
    df = create_dataframe()
    result = extract_columns(df)
    print(result)
